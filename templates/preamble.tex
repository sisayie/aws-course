% This macro is redefined right before each \section (=chapter)
% to hold that chapter's list of section names.
\newcommand{\currentchaptersections}{}

% Tracks which section (subsection) we're currently in, within
% the current chapter. Reset to 0 at each chapter, incremented
% at each section. (We track by index rather than comparing
% against \insertsubsectionhead, since that macro is wrapped in
% a \hyperlink{...}{...} call and won't string-compare cleanly.)
\newcounter{currentsectionindex}

% Draws the list of sections for the current chapter, bolding
% and coloring whichever one matches \thecurrentsectionindex.
% #1 = a font-size command, e.g. \small or \Large
\newcommand{\drawsectionlist}[1]{%
  \begingroup
  #1
  \begin{flushleft}
  \foreach \sec [count=\i] in \currentchaptersections {%
    \ifnum\i=\value{currentsectionindex}%
      \textcolor{structure}{\textbf{\sec}}%
    \else
      \sec%
    \fi
    \par\vspace{0.em}%
  }%
  \end{flushleft}
  \endgroup
}

  \stepcounter{currentsectionindex}%
  \begin{frame}[plain]
